      {Overdue}{Resolve the review comments}
    \kanbantask[author=Bob,due=07/18/2026]
      {Due now}{Complete the current milestone}
    % ISO input is normalized to the selected display format.
    \kanbantask[author=Carol,deadline=2099-12-31]
      {Upcoming}{Prepare the next milestone}
  \end{kanban}}
